# -------------------------------------------------------------------
# LAMMPS Script: Copper Tensile Deformation (FCC)
# Designed for Materials Informatics Project Review
# -------------------------------------------------------------------

# 1. INITIALIZATION
clear
units           metal
dimension       3
boundary        p p p
atom_style      atomic

# Test temperature in K. Defaults to 900; pick another on the command line,
# e.g. "lmp -var T 300 -in in.cu_tensile.txt". The output files are named
# after it (stress_strain_${T}K.txt, dump_${T}K.lammpstrj).
variable        T index 900

# 2. ATOM DEFINITION (Create a 10x10x10 Copper block)
lattice         fcc 3.615
region          box block 0 10 0 10 0 10
create_box      1 box
create_atoms    1 box

# 3. INTERATOMIC POTENTIAL (EAM for Copper)
pair_style      eam/alloy
# NOTE: You must have a Cu EAM file in your folder for this to work!
pair_coeff      * * Cu01.eam.alloy Cu

# 4. EQUILIBRATION (Relax the box at ${T} K)
velocity        all create ${T} 12345 dist gaussian
fix             1 all npt temp ${T} ${T} 0.1 iso 0.0 0.0 1.0
thermo          1000
run             5000
unfix           1

# 5. SETUP DEFORMATION VARIABLES
# Store initial length in Z direction. An equal-style variable is
# re-evaluated every time it is used, so "variable L0 equal lz" would track
# the current length and the strain below would always be 0. Copy the value
# once with ${...} so L0 stays fixed at the start-of-deformation length.
variable        tmp equal lz
variable        L0 equal ${tmp}
print           "Initial length L0 = ${L0} Angstrom"
# Calculate Engineering Strain: (L - L0) / L0
variable        strain equal "(lz - v_L0)/v_L0"
# Calculate Stress in Z direction (Convert from bar to GPa)
variable        stress_GPa equal "-pzz/10000"

# 6. DEFORMATION PROCESS
# Keep X and Y pressure at 0, hold the temperature at ${T} K
fix             1 all npt temp ${T} ${T} 0.1 x 0.0 0.0 1.0 y 0.0 0.0 1.0
# Stretch the box in Z direction at a constant rate
fix             2 all deform 1 z erate 0.01 units box remap x

# 7. DATA OUTPUTS (For OVITO and Graphing)
# Compute the stress on every single atom
compute         atom_stress all stress/atom NULL

# Output 1: Dump file for OVITO (Now includes Z-direction atomic stress!)
dump            1 all custom 500 dump_${T}K.lammpstrj id type x y z c_atom_stress[3]

# Output 2: Text file for the Stress-Strain Curve (Excel/Python)
fix             def_print all print 100 "${strain} ${stress_GPa}" file stress_strain_${T}K.txt screen no

# 8. RUN THE STRETCHING
thermo          500
run             25000

print "SIMULATION COMPLETE! Check your folder for dump_${T}K.lammpstrj and stress_strain_${T}K.txt"
